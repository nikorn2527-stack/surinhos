   167→function formatDate(dateStr: string | null) {
   168→  if (!dateStr) return '-'
   169→  const d = new Date(dateStr)
   170→  return d.toLocaleDateString('th-TH', {
   171→    year: 'numeric',
   172→    month: 'short',
   173→    day: 'numeric',
   174→    hour: '2-digit',
   175→    minute: '2-digit',
   176→  })
   177→}
   178→
   179→function formatCurrency(amount: number | null) {
   180→  if (amount == null) return '-'
   181→  return new Intl.NumberFormat('th-TH', {
   182→    style: 'currency',
   183→    currency: 'THB',
   184→    minimumFractionDigits: 0,
   185→    maximumFractionDigits: 2,
   186→  }).format(amount)
   187→}
   188→
   189→function getTimelineIndex(status: string): number {
   190→  if (status === 'cancelled') return -1
   191→  return statusTimeline.indexOf(status)
   192→}
