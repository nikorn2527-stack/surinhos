    18→// ================= 💡 1. ด่านตรวจความปลอดภัย & ป้องกัน Login ซ้อน ================= //
    19→const verifyToken = (req, res, next) => {
    20→    // รับ Token จาก Header ที่ Frontend ส่งมา
    21→    const authHeader = req.headers['authorization'];
    22→    const token = authHeader && authHeader.split(' ')[1];
    23→
    24→    if (!token) return res.status(401).json({ error: 'กรุณาเข้าสู่ระบบ', forceLogout: true });
    25→
    26→    // ตรวจสอบความถูกต้องและวันหมดอายุของ Token
    27→    jwt.verify(token, SECRET_KEY, (err, decoded) => {
    28→        if (err) return res.status(403).json({ error: 'เซสชันหมดอายุ', forceLogout: true });
    29→
    30→        // ตรวจสอบกับฐานข้อมูลว่า Token ตรงกับของล่าสุดหรือไม่ (ป้องกัน Login ซ้อน)
    31→        db.query('SELECT current_token FROM users WHERE id = ?', [decoded.id], (dbErr, results) => {
    32→            if (dbErr || results.length === 0) return res.status(500).json({ error: 'เกิดข้อผิดพลาดของเซิร์ฟเวอร์' });
    33→
    34→            const latestToken = results[0].current_token;
    35→            if (latestToken !== token) {
    36→                return res.status(401).json({ error: 'มีการเข้าสู่ระบบจากอุปกรณ์อื่น ระบบจะนำคุณออก', forceLogout: true });
    37→            }
    38→
    39→            req.user = decoded; // ให้ข้อมูล user แนบไปกับ request เพื่อให้ API อื่นๆ นำไปใช้ต่อได้
    40→            next(); // ผ่านด่านได้
    41→        });
    42→    });
    43→};
