   168→   168→function formatDate(dateStr: string | null) {
   169→   169→  if (!dateStr) return '-'
   170→   170→  const d = new Date(dateStr)
   171→   171→  return d.toLocaleDateString('th-TH', {
   172→   172→    year: 'numeric',
   173→   173→    month: 'short',
   174→   174→    day: 'numeric',
   175→   175→    hour: '2-digit',
   176→   176→    minute: '2-digit',
   177→   177→  })
   178→   178→}
   179→   179→
   180→   180→function formatCurrency(amount: number | null) {
   181→   181→  if (amount == null) return '-'
   182→   182→  return new Intl.NumberFormat('th-TH', {
   183→   183→    style: 'currency',
   184→   184→    currency: 'THB',
   185→   185→    minimumFractionDigits: 0,
   186→   186→    maximumFractionDigits: 2,
   187→   187→  }).format(amount)
   188→   188→}

   190→   190→function getTimelineIndex(status: string): number {
   191→   191→  if (status === 'cancelled') return -1
   192→   192→  return statusTimeline.indexOf(status)
   193→   193→}
