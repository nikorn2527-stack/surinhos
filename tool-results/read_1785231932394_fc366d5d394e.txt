   166→// ==================== HELPER ====================

   168→function formatDate(dateStr: string | null) {
   169→  if (!dateStr) return '-'
   170→  const d = new Date(dateStr)
   171→  return d.toLocaleDateString('th-TH', {
   172→    year: 'numeric',
   173→    month: 'short',
   174→    day: 'numeric',
   175→    hour: '2-digit',
   176→    minute: '2-digit',
   177→  })
   178→}
   179→
   180→function formatCurrency(amount: number | null) {
   181→  if (amount == null) return '-'
   182→  return new Intl.NumberFormat('th-TH', {
   183→    style: 'currency',
   184→    currency: 'THB',
   185→    minimumFractionDigits: 0,
   186→    maximumFractionDigits: 2,
   187→  }).format(amount)
   188→}
   189→
   190→function getTimelineIndex(status: string): number {
   191→  if (status === 'cancelled') return -1
   192→  return statusTimeline.indexOf(status)
   193→}
