   643→    if (t.status === 'in_progress') msg = '🔧 กำลังซ่อมอุปกรณ์ของคุณอยู่';
   644→    if (t.status === 'returned') msg = '✅ ซ่อมเสร็จแล้ว กรุณารับอุปกรณ์คืน';
   645→    if (t.status === 'closed') msg = '🎉 ดำเนินการเสร็จสิ้นแล้ว';
   646→    if (t.status === 'cancelled') msg = '❌ รายการนี้ถูกยกเลิก';
   647→    if (msg) {
   648→      html += `
   649→        <div class="info-box ${t.status === 'cancelled' ? 'bg-red-50 border-red-100' : 'bg-emerald-50 border-emerald-100'}">
   650→          <p class="text-xs text-gray-600">${msg}</p>
   651→        </div>`;
   652→    }
   653→  }
   654→
   655→  return html;
   656→}
