    19→    19→const verifyToken = (req, res, next) => {

    27→    27→    jwt.verify(token, SECRET_KEY, (err, decoded) => {
    28→    28→        if (err) return res.status(403).json({ error: 'เซสชันหมดอายุ', forceLogout: true });
    29→    29→
    30→    30→        // ตรวจสอบกับฐานข้อมูลว่า Token ตรงกับของล่าสุดหรือไม่ (ป้องกัน Login ซ้อน)

    32→    32→            if (dbErr || results.length === 0) return res.status(500).json({ error: 'เกิดข้อผิดพลาดของเซิร์ฟเวอร์' });
    33→    33→
    34→    34→            const latestToken = results[0].current_token;
